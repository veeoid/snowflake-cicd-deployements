# Readme

## -- Run ONCE PER ENVIRONMENT in Snowsight. Replace <ENV> with DEV, TST, or PRD.

-- Naming convention: MY*PROJECT*<ENV>\_<LAYER>\_DB

CREATE DATABASE IF NOT EXISTS MY*PROJECT*<ENV>_MD_DB;
CREATE SCHEMA IF NOT EXISTS MY_PROJECT_<ENV>\_MD_DB.CATALOG;

CREATE TABLE IF NOT EXISTS MY*PROJECT*<ENV>\_MD_DB.CATALOG.DEPLOYMENT_HISTORY (
OBJECT_NAME VARCHAR,
OBJECT_TYPE VARCHAR,
FILE_PATH VARCHAR,
FILE_HASH VARCHAR,
GIT_SHA VARCHAR,
ENV VARCHAR,
ACTION VARCHAR,
EXECUTED_AT TIMESTAMP_NTZ DEFAULT CURRENT_TIMESTAMP(),
EXECUTED_BY VARCHAR DEFAULT CURRENT_USER()
);

-- Object databases and schemas (the pipeline does NOT create these)
CREATE DATABASE IF NOT EXISTS MY*PROJECT*<ENV>_PREP_DB;
CREATE SCHEMA IF NOT EXISTS MY_PROJECT_<ENV>\_PREP_DB.BASE_MODEL;

CREATE DATABASE IF NOT EXISTS MY*PROJECT*<ENV>_ANALYTICS_DB;
CREATE SCHEMA IF NOT EXISTS MY_PROJECT_<ENV>\_ANALYTICS_DB.INSIGHT_REPORT;
